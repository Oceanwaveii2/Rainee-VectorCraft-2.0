cmake_minimum_required(VERSION 3.20)
# The author uses CMAKE_PROJECT_NAME to construct libstripmaker's target name.
project(stripmaker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT DEFINED SCIP_SDK OR NOT DEFINED STRIPMAKER_SOURCE)
  message(FATAL_ERROR "Pass SCIP_SDK and STRIPMAKER_SOURCE")
endif()
add_library(rainee_scip STATIC backend.cpp)
target_include_directories(rainee_scip PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${SCIP_SDK}/include")
target_link_libraries(rainee_scip PUBLIC "${SCIP_SDK}/lib/libscip.lib")
find_package(osqp CONFIG REQUIRED PATHS "${OSQP_PREFIX}/lib/cmake/osqp" NO_DEFAULT_PATH)
target_link_libraries(rainee_scip PUBLIC osqp::osqpstatic)
add_executable(scip-adapter-tests tests.cpp)
target_link_libraries(scip-adapter-tests PRIVATE rainee_scip)
enable_testing()
add_test(NAME scip-adapter COMMAND scip-adapter-tests)
# Reuse the original project without rewriting its source or classifiers.
# Its historical FindGUROBI names point only to our independent adapter + SCIP.
set(GUROBI_DIR "${SCIP_SDK}" CACHE STRING "SCIP compatibility route" FORCE)
set(GUROBI_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Narrow independent adapter" FORCE)
set(GUROBI_LIBRARY "${SCIP_SDK}/lib/libscip.lib" CACHE FILEPATH "SCIP only" FORCE)
set(GUROBI_CXX_LIBRARY "${SCIP_SDK}/lib/libscip.lib" CACHE FILEPATH "SCIP only" FORCE)
set(GUROBI_CXX_DEBUG_LIBRARY "${SCIP_SDK}/lib/libscip.lib" CACHE FILEPATH "Release research only" FORCE)
add_subdirectory("${STRIPMAKER_SOURCE}" author EXCLUDE_FROM_ALL)
target_link_libraries(rainee_scip PRIVATE Eigen3::Eigen)
get_target_property(author_sources libstripmaker SOURCES)
list(FILTER author_sources EXCLUDE REGEX "stroke_strip_src/Context.cpp$")
set_property(TARGET libstripmaker PROPERTY SOURCES "${author_sources}")
target_sources(libstripmaker PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/context.cpp")
target_include_directories(libstripmaker PRIVATE "${STRIPMAKER_SOURCE}/stroke_strip_src")
target_link_libraries(libstripmaker PUBLIC rainee_scip)
foreach(pair IN ITEMS "clustering-solve|MainSolve" "clustering-endend|MainEndend" "strokestrip|MainStrokeStrip")
  string(REPLACE "|" ";" parts "${pair}")
  list(GET parts 0 target)
  list(GET parts 1 source)
  set_property(SOURCE "${STRIPMAKER_SOURCE}/src/${source}.cpp" TARGET_DIRECTORY ${target}
    APPEND PROPERTY COMPILE_DEFINITIONS main=rainee_author_main)
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
endforeach()
add_custom_target(scip-pipeline DEPENDS clustering-solve clustering-endend strokestrip)
add_executable(junction-worker junction-worker.cpp)
target_link_libraries(junction-worker PRIVATE libstripmaker)
